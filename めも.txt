//正規化
function normalize(str) {
  return str
    .toLowerCase()
    .normalize("NFKC")      // 全角・半角統一
    .replace(/[（）()]/g, "") // 括弧除去
    .replace(/[・,、]/g, " ") // 区切り統一
    .replace(/\s+/g, " ")   // 連続空白削除
    .trim();
}
function kataToHira(str) {
  return str.replace(/[\u30a1-\u30f6]/g, ch =>
    String.fromCharCode(ch.charCodeAt(0) - 0x60)
  );
}
function normalizeJP(str) {
  return normalize(kataToHira(str));
}


//中点削除
function splitMeaning(text) {
  return text.split("・").map(s => s.trim());
}

//かっこ内削除
function removeOptional(text) {
  return text.replace(/（.*?）/g, "");
  //正規表現のひとくちメモ
  //.→すべての文字、*→直前の要素が0回以上繰り返される、?→最初に見つかった)で止まる
  //replace(/[（(].*?[）)]/g, "")←こっちのがいいかも
}

function renderAnswerInputs(meanings) {
  const area = document.getElementById("answer-area");
  area.innerHTML = "";

  const tpl = document.getElementById("answer-template");

  meanings.forEach(m => {
    const node = tpl.content.cloneNode(true);

    node.querySelector(".meaning-no").textContent =
      `${m.no}.`;

    node.querySelector(".answer-input").dataset.meaningNo = m.no;

    area.appendChild(node);
  });
}

function judgeAll(input, meanings) {
  const inputN = normalizeJP(input);

  return meanings.map(m => {
    const ors = removeOptional(m.text)
      .split("・")
      .map(s => normalize(s));

    const level =
      ors.some(o => inputN === o) ? "perfect" :
      ors.some(o => inputN.includes(o)) ? "ok" :
      ors.some(m => m.includes(normalizedInput)) ? "bad" :

  const match = sentence.match(/"(.*?)"/);
  if (!match) return null;

  return {
    answer: match[1],
    question: sentence.replace(/".*?"/, "____")
  };
